function load() {
    try {
        state.products = JSON.parse(localStorage.getItem(K.products) || '[]');
        state.customers = JSON.parse(localStorage.getItem(K.customers) || '[]');
        state.orders = JSON.parse(localStorage.getItem(K.orders) || '[]');
        state.expenses = JSON.parse(localStorage.getItem(K.expenses) || '[]');
        state.debts = JSON.parse(localStorage.getItem(K.debts) || '[]');
        state.settings = Object.assign({}, state.settings, JSON.parse(localStorage.getItem(K.settings) || '{}'));
        state.session = JSON.parse(localStorage.getItem(K.session) || 'null');
    } catch (e) { console.error(e); }
}

function save(key, data) {
    try {
        localStorage.setItem(key, JSON.stringify(data));
        if (typeof autoBackupOnChange === 'function') autoBackupOnChange();
        
        // ☁️ حفظ في السحابة
        if (typeof firebaseReady !== 'undefined' && firebaseReady) {
            const map = {
                'td_products': 'products',
                'td_customers': 'customers',
                'td_expenses': 'expenses',
                'td_debts': 'debts',
                'td_settings': 'settings'
            };
            const coll = map[key];
            if (coll && typeof saveToCloud === 'function') {
                saveToCloud(coll, data);
            }
        }
    } catch (e) { console.error(e); }
}
window.save = save;

function toast(msg, type) {
    type = type || 'success';
    const t = document.getElementById('toast');

    t.textContent = msg;

    clearTimeout(t._timer);
    t._timer = setTimeout(function () { t.classList.remove('show'); }, 2000);

function updateAuthUI() {
    const titleEl = document.getElementById('auth-title');
    const submitEl = document.getElementById('auth-submit');
    const nameG = document.getElementById('auth-name-group');
    const phoneG = document.getElementById('auth-phone-group');
    const addrG = document.getElementById('auth-address-group');
    const passL = document.getElementById('auth-pass-label');
    const box = document.getElementById('auth-switch-box');
    const forgotBox = document.getElementById('forgot-box');
    if (!titleEl) return;

    const t = state.authType, m = state.authMode;

    titleEl.textContent = t === 'admin'
        ? '🔐 دخول المسؤول'
        : (m === 'login' ? 'تسجيل الدخول' : 'إنشاء حساب');
    submitEl.textContent = m === 'login' ? 'دخول' : 'إنشاء الحساب';

    if (nameG) nameG.style.display = (t === 'customer' && m === 'register') ? 'block' : 'none';
    if (phoneG) phoneG.style.display = t === 'customer' ? 'block' : 'none';
    if (addrG) addrG.style.display = (t === 'customer' && m === 'register') ? 'block' : 'none';
    if (passL) passL.textContent = t === 'admin' ? 'كلمة سر المسؤول' : 'كلمة السر';

    if (box) {
        if (t === 'admin') {
            box.style.display = 'none';
        } else {
            box.style.display = 'block';
            document.getElementById('auth-switch-text').textContent =
                m === 'login' ? 'ليس لديك حساب؟' : 'لديك حساب؟';
            document.getElementById('auth-switch-link').textContent =
                m === 'login' ? 'إنشاء حساب' : 'تسجيل الدخول';
        }
    }

    if (forgotBox) {
        forgotBox.style.display = (t === 'customer' && m === 'login') ? 'block' : 'none';
    }
}
window.updateAuthUI = updateAuthUI;

function toggleAuthMode() {
    state.authMode = state.authMode === 'login' ? 'register' : 'login';

function verifyForgotPhone() {
    const phone = document.getElementById('forgot-phone').value.trim();
    if (!phone) { toast('أدخل رقم الهاتف', 'error'); return; }

    const c = state.customers.find(function (x) { return x.phone === phone; });
    if (!c) { toast('رقم الهاتف غير مسجل', 'error'); return; }

    forgotUserId = c.id;
    document.getElementById('forgot-user-info').textContent = '👤 ' + c.name + ' • 📞 ' + c.phone;
    document.getElementById('forgot-step1').style.display = 'none';
    document.getElementById('forgot-step2').style.display = 'block';
}
window.verifyForgotPhone = verifyForgotPhone;

function resetForgotPassword() {
    const pass = document.getElementById('forgot-new-pass').value.trim();
    const confirm = document.getElementById('forgot-confirm-pass').value.trim();

    if (!pass || pass.length < 4) { toast('كلمة السر قصيرة (4 أحرف على الأقل)', 'error'); return; }
    if (pass !== confirm) { toast('كلمتا السر غير متطابقتين', 'error'); return; }

    const c = state.customers.find(function (x) { return x.id === forgotUserId; });
    if (!c) { toast('خطأ، حاول مرة أخرى', 'error'); return; }

    c.password = pass;
    save(K.customers, state.customers);
    closeModal('m-forgot');
    toast('✅ تم تغيير كلمة السر!');

    document.getElementById('auth-phone').value = c.phone;
    document.getElementById('auth-pass').value = '';
    document.getElementById('auth-pass').focus();
}
window.resetForgotPassword = resetForgotPassword;

function backToForgotStep1() {

    document.getElementById('forgot-step2').style.display = 'none';

}

function compressImage(dataUrl, maxSize, callback) {
    const img = new Image();
    img.onload = function () {
        const canvas = document.createElement('canvas');
        let w = img.width, h = img.height;
        if (w > h) { if (w > maxSize) { h = h * maxSize / w; w = maxSize; } }
        else { if (h > maxSize) { w = w * maxSize / h; h = maxSize; } }
        canvas.width = w;
        canvas.height = h;
        canvas.getContext('2d').drawImage(img, 0, 0, w, h);
        callback(canvas.toDataURL('image/jpeg', 0.7));
    };
    img.src = dataUrl;
}
window.compressImage = compressImage;

function getStockStatus(p) {
    if (p.qty === 0) return { cls: 'out', label: 'نفد' };
    if (p.qty <= state.settings.lowStock) return { cls: 'low', label: 'منخفض' };
    return { cls: 'in', label: 'متوفر' };
}
window.getStockStatus = getStockStatus;

function renderAdminProducts() {
    const c = document.getElementById('a-products-list');
    if (!c) return;
    const s = document.getElementById('prod-search');
    const q = s ? s.value.toLowerCase().trim() : '';
    let list = state.products.slice();
    if (q) {
        list = list.filter(function (p) {
            return p.name.toLowerCase().indexOf(q) !== -1 ||
                (p.category && p.category.toLowerCase().indexOf(q) !== -1);
        });
    }
    if (list.length === 0) {
        c.innerHTML = empty('📦', q ? 'لا توجد نتائج' : 'لا توجد منتجات');
        return;
    }
    let html = '';
    list.forEach(function (p) {
        const st = getStockStatus(p);
        const img = p.images && p.images[0];
        const style = img ? 'background-image:url(' + img + ')' : '';
        const cnt = p.images && p.images.length > 1 ? ' • 📸 ' + p.images.length : '';

        let stockLabel = '';
        if (p.qty === 0) stockLabel = '❌ نفد';
        else if (p.qty <= state.settings.lowStock) stockLabel = '⚠️ ' + p.qty;
        else stockLabel = '📦 ' + p.qty;

        html += '<div class="item-card">';
        html += '<div style="display:flex;gap:10px">';
        html += '<div style="width:80px;height:80px;border-radius:12px;overflow:hidden;flex-shrink:0;background:var(--primary-light);display:flex;align-items:center;justify-content:center;font-size:1.8rem;background-size:cover;background-position:center;position:relative;' + style + '">';
        if (!img) html += '🖼️';
        html += '<div class="stock-badge-overlay ' + st.cls + '">' + stockLabel + '</div>';
        html += '</div>';
        html += '<div style="flex:1;min-width:0">';
        html += '<div class="card-top"><div style="flex:1;min-width:0"><h4>' + esc(p.name) + '</h4>';
        html += '<span class="subtitle">' + esc(p.category || '') + cnt + '</span></div>';
        html += '<span class="badge ' + st.cls + '">' + st.label + '</span></div>';
        html += '<div class="card-info">';
        html += '<span class="stock-tag-big ' + st.cls + '">📦 المخزون: ' + p.qty + '</span>';
        html += '<span class="info-tag">💰 ' + formatMoney(p.price) + '</span>';
        html += '</div></div></div>';
        html += '<div class="card-actions" style="margin-top:8px">';
        html += '<button class="btn-mini edit" onclick="quickRestock(' + p.id + ')">📥 + كمية</button>';

function renderStock() {

function downloadPDF() {
    const el = document.getElementById('invoice-printable');
    if (!el) return;
    if (typeof html2canvas === 'undefined' || typeof window.jspdf === 'undefined') {
        toast('المكتبات لم تُحمّل بعد', 'error');
        return;
    }
    toast('⏳ جاري تحضير PDF...');
    html2canvas(el, { scale: 2, backgroundColor: '#ffffff', useCORS: true, logging: false })
        .then(function (canvas) {
            const imgData = canvas.toDataURL('image/png');
            const jsPDF = window.jspdf.jsPDF;
            const pdf = new jsPDF('p', 'mm', 'a4');
            const pw = pdf.internal.pageSize.getWidth();
            const ph = pdf.internal.pageSize.getHeight();
            const m = 10, iw = pw - m * 2;
            const ih = (canvas.height * iw) / canvas.width;
            let left = ih, pos = m;
            pdf.addImage(imgData, 'PNG', m, pos, iw, ih);
            left -= (ph - m * 2);
            while (left > 0) {
                pos = m - (ih - left);
                pdf.addPage();
                pdf.addImage(imgData, 'PNG', m, pos, iw, ih);
                left -= (ph - m * 2);
            }
            pdf.save('Facture-' + document.getElementById('inv-num').textContent + '.pdf');
            toast('✅ تم التحميل');
        })
        .catch(function (err) { console.error(err); toast('خطأ في PDF', 'error'); });
}
window.downloadPDF = downloadPDF;

function createAutoBackup() {
    try {
        const backup = {
            products: state.products, customers: state.customers, orders: state.orders,
            expenses: state.expenses, debts: state.debts, settings: state.settings,
            timestamp: new Date().toISOString()
        };
        localStorage.setItem(BACKUP_KEY, JSON.stringify(backup));
    } catch (e) { console.error(e); }
}
window.createAutoBackup = createAutoBackup;

function autoBackupOnChange() {
    clearTimeout(window._backupTimer);
    window._backupTimer = setTimeout(createAutoBackup, 2000);
}
window.autoBackupOnChange = autoBackupOnChange;

function hasBackup() { return !!localStorage.getItem(BACKUP_KEY); }
window.hasBackup = hasBackup;

function restoreFromBackup() {
    try {
        const data = localStorage.getItem(BACKUP_KEY);
        if (!data) { toast('لا توجد نسخة احتياطية', 'error'); return; }
        const backup = JSON.parse(data);
        const date = formatDateTime(backup.timestamp);
        if (!confirm('استعادة البيانات من نسخة ' + date + '؟')) return;
        if (!confirm('سيتم استبدال البيانات الحالية. متأكد؟')) return;
        if (backup.products) { state.products = backup.products; save(K.products, state.products); }
        if (backup.customers) { state.customers = backup.customers; save(K.customers, state.customers); }
        if (backup.orders) { state.orders = backup.orders; save(K.orders, state.orders); }
        if (backup.expenses) { state.expenses = backup.expenses; save(K.expenses, state.expenses); }
        if (backup.debts) { state.debts = backup.debts; save(K.debts, state.debts); }
        if (backup.settings) { state.settings = backup.settings; save(K.settings, state.settings); }
        toast('✅ تمت الاستعادة! جاري التحديث...');
        setTimeout(function () { location.reload(); }, 1200);
    } catch (e) { console.error(e); toast('خطأ في الاستعادة', 'error'); }
}
window.restoreFromBackup = restoreFromBackup;

function renderShop() {
    const c = document.getElementById('s-products');
    if (!c) return;
    const s = document.getElementById('shop-search');
    const q = s ? s.value.toLowerCase().trim() : '';
    let list = state.products.filter(function (p) { return p.qty > 0; });
    if (q) {
        list = list.filter(function (p) {
            return p.name.toLowerCase().indexOf(q) !== -1 ||
                (p.category && p.category.toLowerCase().indexOf(q) !== -1);
        });
    }
    if (list.length === 0) {
        c.innerHTML = '<div style="grid-column:1/-1">' + empty('🛍️', 'لا توجد منتجات متوفرة') + '</div>';
        return;
    }
    let html = '';
    list.forEach(function (p) {
        const img = p.images && p.images[0];
        const cnt = p.images ? p.images.length : 0;
        const style = img ? 'background-image:url(' + img + ');font-size:0' : '';

        let stockClass = 'in';
        let stockLabel = '📦 ' + p.qty;
        if (p.qty <= state.settings.lowStock) {
            stockClass = 'low';
            stockLabel = '⚠️ ' + p.qty;
        }

        html += '<div class="shop-card" onclick="showProductView(' + p.id + ')">';
        html += '<div class="shop-img" style="position:relative;' + style + '">';
        if (!img) html += '🖼️';
        html += '<div class="stock-badge-overlay ' + stockClass + '">' + stockLabel + '</div>';
        if (cnt > 1) html += '<span class="images-count">📸 ' + cnt + '</span>';
        html += '</div>';
        html += '<div class="shop-body">';
        html += '<h4>' + esc(p.name) + '</h4>';
        html += '<div class="price">' + formatMoney(p.price) + '</div>';
        html += '<div class="stock-info">📦 متوفر: ' + p.qty + '</div>';
        html += '<button class="btn-add-cart" onclick="event.stopPropagation();addToCart(' + p.id + ')">🛒 أضف للسلة</button>';
        html += '</div></div>';
    });
    c.innerHTML = html;
}
window.renderShop = renderShop;

function showProductView(id) {
    const p = state.products.find(function (x) { return x.id === id; });
    if (!p) return;
    state.viewingProduct = p;
    state.viewingQty = 1;

    document.getElementById('pv-name').textContent = p.name;
    document.getElementById('pv-price').textContent = formatMoney(p.price);
    document.getElementById('pv-desc').textContent = p.desc || 'لا يوجد وصف';
    document.getElementById('pv-cat').textContent = '📁 ' + (p.category || 'عام');
    document.getElementById('pv-stock').textContent = '📦 متوفر: ' + p.qty;
    document.getElementById('pv-qty').textContent = '1';

    const mainImg = document.getElementById('pv-main-image');
    const thumbs = document.getElementById('pv-thumbnails');

    if (p.images && p.images.length > 0) {
        mainImg.style.backgroundImage = 'url(' + p.images[0] + ')';
        mainImg.textContent = '';
        if (p.images.length > 1) {
            let th = '';
            p.images.forEach(function (img, i) {
                th += '<div class="pv-thumb ' + (i === 0 ? 'active' : '') + '" style="background-image:url(' + img + ')" onclick="switchProductImage(' + i + ', this)"></div>';
            });
            thumbs.innerHTML = th;
        } else {
            thumbs.innerHTML = '';
        }
    } else {
        mainImg.style.backgroundImage = '';
        mainImg.textContent = '🖼️';
        thumbs.innerHTML = '';
    }
    openModal('m-product-view');
}
window.showProductView = showProductView;

function switchProductImage(index, el) {

    const mainImg = document.getElementById('pv-main-image');

    mainImg.textContent = '';
    document.querySelectorAll('.pv-thumb').forEach(function (t) { t.classList.remove('active'); });

window.switchProductImage = switchProductImage;

function pvChangeQty(delta) {
    const p = state.viewingProduct;
    if (!p) return;
    const newQty = state.viewingQty + delta;
    if (newQty < 1) return;
    if (newQty > p.qty) { toast('الكمية غير متوفرة', 'error'); return; }
    state.viewingQty = newQty;
    document.getElementById('pv-qty').textContent = newQty;
}
window.pvChangeQty = pvChangeQty;

function addToCart(id) {
    const p = state.products.find(function (x) { return x.id === id; });
    if (!p) return;
    const item = state.cart.find(function (c) { return c.id === id; });
    if (item) {
        if (item.qty >= p.qty) { toast('الكمية غير كافية', 'error'); return; }
        item.qty++;
    } else {
        state.cart.push({ id: p.id, name: p.name, price: p.price, qty: 1 });
    }
    updateCartBadge();
    toast('✓ أضيف للسلة');
}
window.addToCart = addToCart;

function updateCartBadge() {
    const badge = document.getElementById('cart-badge');
    if (!badge) return;
    const total = state.cart.reduce(function (s, c) { return s + c.qty; }, 0);
    badge.textContent = total;
    badge.style.display = total > 0 ? 'block' : 'none';
}
window.updateCartBadge = updateCartBadge;

function renderCart() {
    const c = document.getElementById('s-cart-items');
    const summary = document.getElementById('s-cart-summary');
    const btn = document.getElementById('s-checkout-btn');
    if (!c) return;
    if (state.cart.length === 0) {
        c.innerHTML = empty('🛒', 'السلة فارغة');
        if (summary) summary.innerHTML = '';
        if (btn) btn.disabled = true;

        html += '<br><small style="color:var(--text-mid);font-size:.72rem">' + formatMoney(item.price) + '</small></div>';
        html += '<div class="qty-controls">';
        html += '<button class="qty-btn" onclick="changeQty(' + item.id + ',-1)">−</button>';
        html += '<span class="qty-num">' + item.qty + '</span>';
        html += '<button class="qty-btn" onclick="changeQty(' + item.id + ',1)">+</button>';

        html += '<strong style="color:var(--primary);font-size:.85rem;min-width:60px;text-align:left">' + formatMoney(item.price * item.qty) + '</strong>';
        html += '</div>';
    });

        summary.innerHTML =
            '<div class="row"><span>عدد المنتجات</span><span>' + state.cart.length + '</span></div>' +

    }
}
window.renderCart = renderCart;

function changeQty(id, delta) {
    const item = state.cart.find(function (c) { return c.id === id; });
    if (!item) return;
    const p = state.products.find(function (x) { return x.id === id; });
    const newQty = item.qty + delta;
    if (newQty <= 0) {
        state.cart = state.cart.filter(function (c) { return c.id !== id; });
    } else if (p && newQty <= p.qty) {
        item.qty = newQty;
    } else {
        toast('الكمية غير متوفرة', 'error');
        return;
    }
    updateCartBadge();
    renderCart();
}
window.changeQty = changeQty;

function getNotifications() {
    const notifs = [];
    const role = state.session ? state.session.role : null;

    if (role === 'admin') {
        const outStock = state.products.filter(function (p) { return p.qty === 0; });
        const lowStock = state.products.filter(function (p) { return p.qty > 0 && p.qty <= state.settings.lowStock; });

        outStock.forEach(function (p) {
            notifs.push({
                id: 'out-' + p.id, type: 'danger',
                title: '❌ المنتج نفد',
                desc: '"' + p.name + '" غير متوفر حالياً',
                time: 'الآن'
            });
        });

        lowStock.forEach(function (p) {
            notifs.push({
                id: 'low-' + p.id, type: 'warning',
                title: '⚠️ مخزون منخفض',
                desc: '"' + p.name + '" — بقي ' + p.qty + ' فقط',
                time: 'الآن'
            });
        });

        const pendingOrders = state.orders.filter(function (o) { return o.status === 'pending'; });
        pendingOrders.forEach(function (o) {
            notifs.push({
                id: 'order-' + o.id, type: 'info',
                title: '🛒 طلبية جديدة',
                desc: o.customerName + ' • ' + formatMoney(o.total),
                time: formatDateTime(o.createdAt),
                onClick: 'viewOrder(' + o.id + ')'
            });
        });

        const unpaidDebts = state.debts.filter(function (d) { return !d.paid; });
        unpaidDebts.forEach(function (d) {
            notifs.push({
                id: 'debt-' + d.id, type: 'warning',
                title: '💳 دين غير مسدد',
                desc: d.name + ' — ' + formatMoney(d.amount),
                time: formatDate(d.date)
            });
        });
    }

    if (role === 'customer') {
        const myOrders = state.orders.filter(function (o) { return o.customerId === state.session.id; });
        myOrders.forEach(function (o) {
            let type = 'info', title = '', desc = '';
            if (o.status === 'confirmed') {
                type = 'success'; title = '✅ تم تأكيد طلبك';
                desc = 'طلب #' + o.id.toString().slice(-6) + ' — ' + formatMoney(o.total);
            } else if (o.status === 'shipped') {
                type = 'info'; title = '🚚 طلبك في الطريق';
                desc = 'طلب #' + o.id.toString().slice(-6) + ' — سيصلك قريباً';
            } else if (o.status === 'delivered') {
                type = 'success'; title = '📦 تم التسليم';
                desc = 'طلب #' + o.id.toString().slice(-6) + ' — شكراً لثقتك';
            } else if (o.status === 'cancelled') {
                type = 'danger'; title = '❌ تم إلغاء طلبك';
                desc = 'طلب #' + o.id.toString().slice(-6);
            } else if (o.status === 'pending') {
                type = 'warning'; title = '🕐 طلب قيد المراجعة';
                desc = 'طلب #' + o.id.toString().slice(-6) + ' — سيتم التواصل معك';
            }
            if (title) {
                notifs.push({
                    id: 'cust-order-' + o.id, type: type,
                    title: title, desc: desc,
                    time: formatDateTime(o.updatedAt || o.createdAt)
                });
            }
        });
    }

    return notifs;
}
window.getNotifications = getNotifications;

function updateNotifBadge() {
    const notifs = getNotifications();
    const badgeId = state.session && state.session.role === 'admin' ? 'admin-notif-badge' : 'cust-notif-badge';
    const badge = document.getElementById(badgeId);
    if (!badge) return;
    const count = notifs.length;
    if (count === 0) {
        badge.style.display = 'none';
    } else {
        badge.style.display = 'flex';
        badge.textContent = count > 99 ? '99+' : count;
    }
}
window.updateNotifBadge = updateNotifBadge;

function renderNotifList() {
    const c = document.getElementById('notif-list');
    if (!c) return;
    const notifs = getNotifications();
    if (notifs.length === 0) {
        c.innerHTML = '<div class="notif-empty"><span class="emoji">🎉</span><p>لا توجد إشعارات</p><p style="font-size:.75rem;margin-top:6px">كل شيء على ما يرام!</p></div>';
        return;
    }
    let html = '';
    notifs.forEach(function (n) {
        const onclick = n.onClick ? 'onclick="closeNotifPanel();setTimeout(function(){' + n.onClick + '},200)"' : '';
        html += '<div class="notif-item ' + n.type + '" ' + onclick + '>';
        html += '<div class="notif-title">' + n.title + '</div>';
        html += '<div class="notif-desc">' + n.desc + '</div>';
        html += '<div class="notif-time">🕐 ' + n.time + '</div>';
        html += '</div>';
    });
    c.innerHTML = html;
}
window.renderNotifList = renderNotifList;

function renderStockAlerts() {
    const home = document.getElementById('a-home');
    if (!home) return;
    const old = home.querySelector('.stock-alert-block');
    if (old) old.remove();

    const outStock = state.products.filter(function (p) { return p.qty === 0; });
    const lowStock = state.products.filter(function (p) { return p.qty > 0 && p.qty <= state.settings.lowStock; });

    if (outStock.length === 0 && lowStock.length === 0 && hasBackup()) return;

    const block = document.createElement('div');
    block.className = 'stock-alert-block';

    if (outStock.length > 0) {
        const names = outStock.slice(0, 3).map(function (p) { return p.name; }).join('، ');
        const more = outStock.length > 3 ? ' و ' + (outStock.length - 3) + ' آخر' : '';
        const div = document.createElement('div');
        div.className = 'stock-alert danger';
        div.onclick = function () { adminGo('stock'); };
        div.innerHTML = '<span style="font-size:1.5rem">❌</span><div><strong>' + outStock.length + ' منتج نفد</strong><br><span style="font-size:.75rem;font-weight:500">' + names + more + '</span></div>';
        block.appendChild(div);
    }

    if (lowStock.length > 0) {
        const names = lowStock.slice(0, 3).map(function (p) { return p.name; }).join('، ');
        const more = lowStock.length > 3 ? ' و ' + (lowStock.length - 3) + ' آخر' : '';
        const div = document.createElement('div');
        div.className = 'stock-alert';
        div.onclick = function () { adminGo('stock'); };
        div.innerHTML = '<span style="font-size:1.5rem">⚠️</span><div><strong>' + lowStock.length + ' منتج منخفض</strong><br><span style="font-size:.75rem;font-weight:500">' + names + more + '</span></div>';
        block.appendChild(div);
    }

    if (!hasBackup() && (state.products.length > 0 || state.orders.length > 0)) {
        const div = document.createElement('div');
        div.className = 'backup-alert';
        div.innerHTML = '<span style="font-size:1.4rem">💾</span><div><strong>لم تُنشئ نسخة احتياطية بعد!</strong><br><span style="font-size:.72rem;font-weight:500">اضغط لحفظ البيانات الآن</span></div>';
        div.onclick = function () {
            createAutoBackup();
            toast('✅ تم إنشاء نسخة احتياطية');
            renderStockAlerts();
        };
        block.appendChild(div);
    }

    home.insertBefore(block, home.firstChild);
}
window.renderStockAlerts = renderStockAlerts;

function setupClock() {
    function updateClock() {
        const now = new Date();
        const h = String(now.getHours()).padStart(2, '0');
        const m = String(now.getMinutes()).padStart(2, '0');
        const s = String(now.getSeconds()).padStart(2, '0');
        const time = h + ':' + m + ':' + s;
        const day = String(now.getDate()).padStart(2, '0');
        const month = String(now.getMonth() + 1).padStart(2, '0');
        const year = now.getFullYear();
        const date = day + '/' + month + '/' + year;

        const wc = document.getElementById('top-clock');
        const wd = document.getElementById('top-date');
        if (wc) wc.textContent = time;
        if (wd) wd.textContent = date;

        const sc = document.getElementById('shop-clock');
        const sd = document.getElementById('shop-date');
        if (sc) sc.textContent = time;
        if (sd) sd.textContent = date;
    }
    updateClock();
    setInterval(updateClock, 1000);
}
window.setupClock = setupClock;

}

function setupAdminClockPress() {
    const clock = document.getElementById('shop-clock');
    if (!clock) { console.warn('shop-clock غير موجود'); return; }

    const LONG_PRESS = 2000;
    let pressTimer = null;
    let isPressing = false;

    function goToAdmin() {
        console.log('🚀 الانتقال لصفحة المسؤول...');
        // جرّب عدة مسارات
        try {
            window.location.href = 'admin.html';
        } catch (e) {
            console.error(e);
        }
        // إذا لم يعمل، جرّب بعد 300ms
        setTimeout(function () {
            if (window.location.pathname.indexOf('admin') === -1) {
                window.location.replace('./admin.html');
            }
        }, 300);
    }

    function startPress(e) {
        if (e.type === 'mousedown' && e.button !== 0) return;
        isPressing = true;
        clock.classList.add('pressing');
        if (navigator.vibrate) navigator.vibrate(20);

        clearTimeout(pressTimer);
        pressTimer = setTimeout(function () {
            if (!isPressing) return;
            isPressing = false;
            clock.classList.remove('pressing');
            clock.classList.add('success');
            if (navigator.vibrate) navigator.vibrate([80, 50, 80, 50, 150]);
            console.log('✅ نجح الضغط المطول - الانتقال...');
            setTimeout(goToAdmin, 300);
        }, LONG_PRESS);
    }

    function cancelPress() {
        isPressing = false;
        clearTimeout(pressTimer);
        clock.classList.remove('pressing');
    }

    clock.addEventListener('touchstart', startPress, { passive: true });
    clock.addEventListener('touchend', cancelPress);
    clock.addEventListener('touchcancel', cancelPress);
    clock.addEventListener('mousedown', startPress);
    clock.addEventListener('mouseup', cancelPress);
    clock.addEventListener('mouseleave', cancelPress);
    clock.addEventListener('contextmenu', function (e) { e.preventDefault(); });

    console.log('✅ الضغط المطول جاهز');
}
window.setupAdminClockPress = setupAdminClockPress;

function setupEmergencyAdminButton() {
    const btn = document.getElementById('admin-emergency');
    if (!btn) { console.log('⚠️ زر الطوارئ غير موجود'); return; }

    let clickCount = 0;
    let clickTimer = null;

    function goToAdminPage() {
        console.log('🚀 الانتقال إلى admin.html...');
        // ثلاث طرق - جرّب واحدة بعد الأخرى
        try { window.location.href = 'admin.html'; } catch (e) { console.error(e); }
        setTimeout(function() { 
            if (window.location.pathname.indexOf('admin') === -1) {
                try { window.location.assign('./admin.html'); } catch (e) { console.error(e); }
            }
        }, 200);
        setTimeout(function() {
            if (window.location.pathname.indexOf('admin') === -1) {
                try { window.location.replace('admin.html'); } catch (e) { console.error(e); }
            }
        }, 500);
    }

    function handleClick(e) {
        e.preventDefault();
        clickCount++;
        if (navigator.vibrate) navigator.vibrate(15);

        clearTimeout(clickTimer);
        clickTimer = setTimeout(function () {
            clickCount = 0;
        }, 2000);

        if (clickCount >= 5) {
            clickCount = 0;
            if (navigator.vibrate) navigator.vibrate([50, 30, 50, 30, 100]);
            goToAdminPage();
        }
    }

    btn.addEventListener('click', handleClick);
    btn.addEventListener('touchend', handleClick);
    console.log('✅ زر الطوارئ جاهز - اضغط 5 مرات على الزاوية السفلية اليسرى');
}
window.setupEmergencyAdminButton = setupEmergencyAdminButton;

async function initCloudSync() {
    if (typeof initFirebase !== 'function') {
        console.warn('⚠️ firebase-config.js غير محمل');
        return;
    }
    
    try {
        await initFirebase();
        
        // 1. تحميل البيانات من السحابة (إن وُجدت)
        const cols = ['products', 'customers', 'expenses', 'debts', 'settings'];
        for (const col of cols) {
            const cloudData = await loadFromCloud(col);
            if (cloudData && Array.isArray(cloudData) && cloudData.length >= 0) {
                // دمج: نأخذ الأحدث من السحابة
                if (col === 'products') { state.products = cloudData; localStorage.setItem(K.products, JSON.stringify(cloudData)); }
                else if (col === 'customers') { state.customers = cloudData; localStorage.setItem(K.customers, JSON.stringify(cloudData)); }
                else if (col === 'expenses') { state.expenses = cloudData; localStorage.setItem(K.expenses, JSON.stringify(cloudData)); }
                else if (col === 'debts') { state.debts = cloudData; localStorage.setItem(K.debts, JSON.stringify(cloudData)); }
                else if (col === 'settings') { state.settings = Object.assign({}, state.settings, cloudData); localStorage.setItem(K.settings, JSON.stringify(state.settings)); }
            } else {
                // لم تكن في السحابة → ارفعها الآن
                if (col === 'products') saveToCloud('products', state.products);
                else if (col === 'customers') saveToCloud('customers', state.customers);
                else if (col === 'expenses') saveToCloud('expenses', state.expenses);
                else if (col === 'debts') saveToCloud('debts', state.debts);
                else if (col === 'settings') saveToCloud('settings', state.settings);
            }
        }
        
        // 2. مراقبة التغييرات في الوقت الحقيقي
        watchCloud('products', function(data) {
            if (Array.isArray(data)) {
                state.products = data;
                localStorage.setItem(K.products, JSON.stringify(data));
                if (typeof renderShop === 'function' && document.getElementById('s-products')) renderShop();
                if (typeof renderAdminProducts === 'function' && document.getElementById('a-products-list')) renderAdminProducts();
                if (typeof renderAdminHome === 'function' && document.getElementById('a-recent-orders')) renderAdminHome();
            }
        });
        
        watchCloud('customers', function(data) {
            if (Array.isArray(data)) {
                state.customers = data;
                localStorage.setItem(K.customers, JSON.stringify(data));
            }
        });
        
        watchCloud('settings', function(data) {
            if (data && typeof data === 'object') {
                state.settings = Object.assign({}, state.settings, data);
                localStorage.setItem(K.settings, JSON.stringify(state.settings));
                if (typeof applyBranding === 'function') applyBranding();
            }
        });
        
        // 3. مراقبة الطلبات (المهم!)
        if (typeof watchOrders === 'function') {
            watchOrders(function(cloudOrders) {
                console.log('📥 استلمت', cloudOrders.length, 'طلبية من السحابة');
                
                // دمج الطلبات السحابية مع المحلية
                const localIds = new Set(state.orders.map(function(o) { return o.id; }));
                cloudOrders.forEach(function(co) {
                    if (!localIds.has(co.id)) {
                        // طلب جديد من السحابة → أضفه
                        state.orders.push(co);
                    } else {
                        // تحديث الطلب الموجود
                        const idx = state.orders.findIndex(function(o) { return o.id === co.id; });
                        if (idx !== -1) {
                            state.orders[idx] = Object.assign({}, state.orders[idx], co);
                        }
                    }
                });
                
                // ترتيب حسب التاريخ
                state.orders.sort(function(a, b) {
                    return new Date(b.createdAt || 0) - new Date(a.createdAt || 0);
                });
                
                localStorage.setItem(K.orders, JSON.stringify(state.orders));
                
                // تحديث الواجهات
                if (typeof renderOrders === 'function' && document.getElementById('a-orders-list')) renderOrders();
                if (typeof renderAdminHome === 'function' && document.getElementById('a-recent-orders')) renderAdminHome();
                if (typeof renderMyOrders === 'function' && document.getElementById('s-orders-list')) renderMyOrders();
                if (typeof updateNotifBadge === 'function') updateNotifBadge();
            });
        }
        
        console.log('✅ Firebase sync جاهز');
        
    } catch (e) {
        console.error('❌ Firebase sync error:', e);
    }
}
window.initCloudSync = initCloudSync;

            if (item) {
                if (item.qty + state.viewingQty > p.qty) { toast('الكمية غير كافية', 'error'); return; }
                item.qty += state.viewingQty;
            } else {
                state.cart.push({ id: p.id, name: p.name, price: p.price, qty: state.viewingQty });
            }
            updateCartBadge();
            closeModal('m-product-view');
            toast('✓ أضيف ' + state.viewingQty + ' للسلة');
        });
    }

    // 13. البحث
    const prodSearch = document.getElementById('prod-search');
    if (prodSearch) prodSearch.addEventListener('input', renderAdminProducts);
    const shopSearch = document.getElementById('shop-search');
    if (shopSearch) shopSearch.addEventListener('input', renderShop);

    // 14. إغلاق النوافذ بالخلفية
    document.querySelectorAll('.modal-overlay').forEach(function (overlay) {
        overlay.addEventListener('click', function (e) {
            if (e.target === overlay) {
                overlay.classList.remove('active');
                document.body.style.overflow = '';
            }
        });
    });

    // 15. استئناف الجلسة (حسب نوع الصفحة)
    if (PAGE_TYPE === 'admin') {
        if (state.session && state.session.role === 'admin') {
            enterAdmin();
        } else {
            if (state.session && state.session.role === 'customer') {
                state.session = null;
                localStorage.removeItem(K.session);
            }
            goScreen('screen-admin');
        }
    } else {
        // صفحة الزبون
        if (state.session && state.session.role === 'customer') {
            enterShop();
        } else if (state.session && state.session.role === 'admin') {
            state.session = null;
            localStorage.removeItem(K.session);
            goScreen('screen-welcome');
        } else {
            goScreen('screen-welcome');
        }
    }

    // 16. تحديث شارة السلة
    updateCartBadge();

    // 17. تحديث الإشعارات
    updateNotifBadge();
  // 🔥 تهيئة Firebase
initCloudSync();

    console.log('✅ Thier Décor جاهز - الصفحة:', PAGE_TYPE);
});
